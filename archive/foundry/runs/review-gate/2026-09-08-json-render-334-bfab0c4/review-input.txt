1: # @json-render/tanstack-start
2: 
3: TanStack Start renderer for [@json-render/core](https://json-render.dev).
4: Define routes, layouts, head metadata, state, and loader-backed pages as JSON,
5: then render them through TanStack Router and Start SSR.
6: 
7: ## Installation
8: 
9: ```bash
10: npm install @json-render/core @json-render/react @json-render/tanstack-start
11: ```
12: 
13: ## Quick Start
14: 
15: ### 1. Define the catalog
16: 
17: Include the definitions for the built-in `Slot` and `Link` components. Their
18: React implementations are added automatically by `PageRenderer`.
19: 
20: ```typescript
21: import { defineCatalog } from "@json-render/core";
22: import {
23:   schema,
24:   startComponentDefinitions,
25: } from "@json-render/tanstack-start/server";
26: 
27: export const catalog = defineCatalog(schema, {
28:   components: {
29:     ...startComponentDefinitions,
30:     Card: cardDefinition,
31:     Container: containerDefinition,
32:     NavBar: navBarDefinition,
33:     Post: postDefinition,
34:   },
35:   actions: {},
36: });
37: ```
38: 
39: ### 2. Define the application
40: 
41: ```typescript
42: import type { StartAppSpec } from "@json-render/tanstack-start";
43: 
44: export const spec: StartAppSpec = {
45:   metadata: {
46:     title: { default: "My App", template: "%s | My App" },
47:   },
48:   layouts: {
49:     main: {
50:       root: "shell",
51:       elements: {
52:         shell: {
53:           type: "Container",
54:           props: {},
55:           children: ["nav", "slot"],
56:         },
57:         nav: { type: "NavBar", props: {}, children: [] },
58:         slot: { type: "Slot", props: {}, children: [] },
59:       },
60:     },
61:   },
62:   routes: {
63:     "/": {
64:       layout: "main",
65:       metadata: { title: "Home" },
66:       page: {
67:         root: "hero",
68:         elements: {
69:           hero: {
70:             type: "Card",
71:             props: { title: "Welcome" },
72:             children: [],
73:           },
74:         },
75:       },
76:     },
77:     "/blog/$slug": {
78:       layout: "main",
79:       loader: "post",
80:       page: {
81:         root: "post",
82:         elements: {
83:           post: {
84:             type: "Post",
85:             props: { post: { $state: "/post" } },
86:             children: [],
87:           },
88:         },
89:       },
90:     },
91:   },
92: };
93: ```
94: 
95: ### 3. Create the route helpers
96: 
97: ```typescript
98: // src/lib/json-app.ts
99: import { createStartApp } from "@json-render/tanstack-start/server";
100: import { spec } from "./spec";
101: 
102: export const { getPageData, getHead, getStaticPaths } = createStartApp({
103:   spec,
104:   loaders: {
105:     post: async ({ slug }) => ({ post: await getPost(slug as string) }),
106:   },
107: });
108: ```
109: 
110: ### 4. Wire a splat route
111: 
112: ```tsx
113: // src/routes/$.tsx
114: import { createFileRoute, notFound } from "@tanstack/react-router";
115: import {
116:   PageRenderer,
117:   StartErrorBoundary,
118:   StartLoading,
119:   StartNotFound,
120: } from "@json-render/tanstack-start";
121: import { getHead, getPageData } from "@/lib/json-app";
122: 
123: export const Route = createFileRoute("/$")({
124:   loader: async ({ location }) => {
125:     const data = await getPageData({ pathname: location.pathname });
126:     if (!data) throw notFound();
127:     return data;
128:   },
129:   head: ({ match }) => getHead({ pathname: match.pathname }),
130:   component: Page,
131:   pendingComponent: StartLoading,
132:   errorComponent: StartErrorBoundary,
133:   notFoundComponent: StartNotFound,
134: });
135: 
136: function Page() {
137:   return <PageRenderer {...Route.useLoaderData()} />;
138: }
139: ```
140: 
141: TanStack Router loaders are isomorphic. If your spec factory or named loaders
142: contain secrets or server-only imports, call `getPageData` and `getHead` from a
143: TanStack Start `createServerFn` and return the resulting data from the route
144: loader.
145: 
146: ### 5. Provide the registry
147: 
148: ```tsx
149: // src/routes/__root.tsx
150: import {
151:   createRootRoute,
152:   HeadContent,
153:   Outlet,
154:   Scripts,
155: } from "@tanstack/react-router";
156: import { StartAppProvider } from "@json-render/tanstack-start";
157: import { registry, handlers } from "@/lib/registry";
158: import { spec } from "@/lib/spec";
159: 
160: export const Route = createRootRoute({ component: Root });
161: 
162: function Root() {
163:   return (
164:     <html lang="en">
165:       <head>
166:         <HeadContent />
167:       </head>
168:       <body>
169:         <StartAppProvider
170:           registry={registry}
171:           handlers={handlers}
172:           spec={spec}
173:         >
174:           <Outlet />
175:         </StartAppProvider>
176:         <Scripts />
177:       </body>
178:     </html>
179:   );
180: }
181: ```
182: 
183: Passing `spec` lets the Router boundary components automatically render the
184: matched route's `loading`, `error`, and `notFound` specs. An explicit
185: `loadingSpec`, `errorSpec`, or `notFoundSpec` prop overrides this lookup. If the
186: application spec is server-only, omit `spec` from the provider and supply those
187: explicit props from client-safe fallback specs.
188: 
189: Pass named functions through the provider when generated props use
190: `$computed`:
191: 
192: ```tsx
193: <StartAppProvider
194:   registry={registry}
195:   spec={spec}
196:   functions={{ uppercase: ({ value }) => String(value).toUpperCase() }}
197: >
198:   <Outlet />
199: </StartAppProvider>
200: ```
201: 
202: ## Route Patterns
203: 
204: | Pattern       | Matches       | Loader params            |
205: | ------------- | ------------- | ------------------------ |
206: | `/`           | `/`           | `{}`                     |
207: | `/about`      | `/about`      | `{}`                     |
208: | `/blog/$slug` | `/blog/hello` | `{ slug: "hello" }`      |
209: | `/docs/$`     | `/docs/a/b`   | `{ _splat: "a/b" }`       |
210: 
211: Static routes are included in `getStaticPaths()`. Dynamic routes are included
212: when their route spec supplies `staticParams`. Loader parameters are URL-decoded,
213: and splat content is a slash-delimited string under `_splat`. Parameter values
214: emitted by `getStaticPaths()` are URL-encoded.
215: 
216: Initial state is merged in this order: application state, layout state, page
217: state, then loader data. Later sources override earlier values.
218: 
219: Map the paths to TanStack Start's top-level `pages` option when prerendering:
220: 
221: ```typescript
222: const pages = (await getStaticPaths()).map((path) => ({ path }));
223: ```
224: 
225: ## Entry Points
226: 
227: | Import                                | Description                                                   |
228: | ------------------------------------- | ------------------------------------------------------------- |
229: | `@json-render/tanstack-start`         | Provider, page renderer, Link, and route fallback components  |
230: | `@json-render/tanstack-start/server`  | App factory, schema, matcher, metadata, and prerender helpers |
231: | `@json-render/tanstack-start/catalog` | Server-safe definitions for built-in Slot and Link components |
232: 
233: See the [full API reference](https://json-render.dev/docs/api/tanstack-start).

8:   type ComponentRenderProps,

18: }

21:   return <>{children}</>;

24: /** Render page data returned by `createStartApp` inside an optional layout. */
25: export function PageRenderer({
26:   spec,
27:   initialState,
28:   layoutSpec,
29:   loading,
30: }: PageRendererProps) {
31:   const {
32:     registry,
33:     handlers,
34:     spec: appSpec,
35:     functions,
36:     navigate,
37:   } = useStartApp();
38:   const renderedPathname = useMatch({
39:     strict: false,
40:     select: (match) => match.pathname,
41:   });
42:   const augmentedRegistry: ComponentRegistry = useMemo(
43:     () => ({ ...registry, Link, Slot }),
44:     [registry],
45:   );
46:   const actionHandlers = useMemo(
47:     () => ({
48:       ...handlers,
49:       navigate: (params: Record<string, unknown>) => {
50:         const href = params.href;
51:         if (typeof href === "string") navigate(href);
52:       },
53:     }),
54:     [handlers, navigate],
55:   );
56:   const resolvedInitialState = useMemo(() => {
57:     if (initialState !== undefined) return initialState;
58:     if (!appSpec?.state && !layoutSpec?.state && !spec.state) return undefined;
59:     return { ...appSpec?.state, ...layoutSpec?.state, ...spec.state };
60:   }, [appSpec?.state, initialState, layoutSpec?.state, spec.state]);
61: 
62:   const page = (
63:     <Renderer spec={spec} registry={augmentedRegistry} loading={loading} />
64:   );
65: 
66:   // Key from the rendered match rather than the global location. During a
67:   // pending navigation, TanStack advances the location while keeping the
68:   // previous match mounted until the next page is ready.
69:   return (
70:     <JSONUIProvider
71:       key={renderedPathname}
72:       registry={augmentedRegistry}
73:       initialState={resolvedInitialState}
74:       handlers={actionHandlers}
75:       navigate={navigate}
76:       functions={functions}
77:     >
78:       {layoutSpec ? (
79:         <LayoutWithSlot
80:           layoutSpec={layoutSpec}
81:           registry={augmentedRegistry}
82:           loading={loading}
83:         >
84:           {page}
85:         </LayoutWithSlot>
86:       ) : (

88:       )}
89:     </JSONUIProvider>
90:   );
91: }

4:   pattern: string;
5:   regex: RegExp;
6:   paramNames: string[];

10: const SPLAT_PARAM = "_splat";
11: const STATIC_SEGMENT_RANK = 3;
12: const DYNAMIC_SEGMENT_RANK = 2;
13: const SPLAT_SEGMENT_RANK = 1;
14: 
15: /** Compile a TanStack Router pattern into a pathname matcher. */
16: function compileRoute(pattern: string): CompiledRoute {
17:   const paramNames: string[] = [];
18:   const segments = pattern === "/" ? [""] : pattern.split("/").slice(1);
19:   const regexParts: string[] = [];
20:   const segmentRanks: number[] = [];
21: 
22:   for (const segment of segments) {
23:     if (segment === "$") {
24:       paramNames.push(SPLAT_PARAM);
25:       segmentRanks.push(SPLAT_SEGMENT_RANK);
26:       regexParts.push("(?:/(.*))?");
27:     } else if (segment.startsWith("$") && segment.length > 1) {
28:       paramNames.push(segment.slice(1));
29:       segmentRanks.push(DYNAMIC_SEGMENT_RANK);
30:       regexParts.push("/([^/]+)");
31:     } else {
32:       segmentRanks.push(STATIC_SEGMENT_RANK);
33:       regexParts.push(`/${escapeRegExp(segment)}`);
34:     }
35:   }
36: 
37:   return {
38:     pattern,
39:     regex: new RegExp(
40:       pattern === "/" ? "^/$" : `^${regexParts.join("")}$`,
41:       "i",
42:     ),
43:     paramNames,
44:     segmentRanks,
45:   };
46: }
47: 
48: function escapeRegExp(value: string): string {
49:   return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
50: }

52: /** Match a pathname using TanStack Router's static, dynamic, and splat forms. */
53: export function matchRoute(
54:   spec: StartAppSpec,
55:   pathname: string,
56: ): MatchedRoute | null {
57:   const normalizedPath = normalizePathname(pathname);
58:   const compiled = Object.keys(spec.routes).map(compileRoute);
59: 
60:   compiled.sort((a, b) => {
61:     const segmentCount = Math.max(a.segmentRanks.length, b.segmentRanks.length);
62:     for (let index = 0; index < segmentCount; index++) {
63:       const aRank = a.segmentRanks[index] ?? 0;
64:       const bRank = b.segmentRanks[index] ?? 0;
65:       if (aRank !== bRank) return bRank - aRank;
66:     }
67:     return 0;
68:   });
69: 
70:   for (const candidate of compiled) {
71:     const match = candidate.regex.exec(normalizedPath);
72:     if (!match) continue;
73: 
74:     const params: Record<string, string> = {};
75:     let validParams = true;
76:     for (let index = 0; index < candidate.paramNames.length; index++) {
77:       const name = candidate.paramNames[index]!;
78:       const value = match[index + 1];
79:       try {
80:         params[name] = decodeURIComponent(value ?? "");
81:       } catch {
82:         validParams = false;
83:         break;
84:       }
85:     }
86:     if (!validParams) continue;
87: 
88:     return {
89:       route: spec.routes[candidate.pattern]!,
90:       pattern: candidate.pattern,
91:       params,
92:     };
93:   }
94: 
95:   return null;
96: }
